function renderUsefulForYear(year){ usefulWrap.innerHTML = ""; const cfg = YEARS[year]; // إذا ما فيه Useful في هذه السنة -> اخفِ القسم if (!cfg?.USEFUL) { usefulWrap.classList.add("hidden"); return; } // إذا موجود -> أظهره usefulWrap.classList.remove("hidden"); const usefulCard = createProjectCard(cfg.USEFUL.TITLE, { isUseful: true, usefulOptions: cfg.USEFUL.OPTIONS || [], usefulLinksMap: cfg.USEFUL.LINKS || {} }); usefulWrap.appendChild(usefulCard); }
